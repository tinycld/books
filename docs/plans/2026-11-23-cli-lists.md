```go
// Package exitcode is the exit-code contract of the tinycld CLI, for the
// shell and for the command modules of packages
// (tinycld.org/packages/<slug>/cli).
//
// main exits with the code of the first *Error in the chain of the error a
// command returns, and with Failed for any other error. A command that wants
// a script or an agent to tell "sign in again" from "the server refused the
// data" returns Classify(err), which reads the class from the error itself.
//
// The numbers are a public contract: scripts test them. Add a class at the
// end; never renumber one.
package exitcode

import (
	"context"
	"errors"
	"net"

	"tinycld.org/cli/client"
)

const (
	// OK: the command did what was asked.
	OK = 0
	// Failed: any failure with no class of its own.
	Failed = 1
	// Usage: the call itself is wrong (flags, arguments, input document).
	// Nothing was sent to the server.
	Usage = 2
	// Auth: there is no usable credential (HTTP 401). Sign in again.
	Auth = 3
	// Forbidden: the server knows the caller and refuses the action
	// (HTTP 403): a missing OAuth scope or a missing permission.
	Forbidden = 4
	// NotFound: the record or route does not exist, or the caller may not
	// see it (HTTP 404).
	NotFound = 5
	// Refused: the server refused the data (any other HTTP 4xx). Change the
	// input; the same call fails again.
	Refused = 6
	// Unavailable: the server did not answer, or answered HTTP 408, 429 or
	// 5xx. The same call can succeed later.
	Unavailable = 7
)

// Error carries the exit code main uses for Err.
type Error struct {
	Code int
	Err  error
}

func (e *Error) Error() string { return e.Err.Error() }
func (e *Error) Unwrap() error { return e.Err }

// New gives err the exit code. A nil err stays nil.
func New(code int, err error) error {
	if err == nil {
		return nil
	}
	return &Error{Code: code, Err: err}
}

// Of is the exit code for err: OK for nil, the code of the first *Error in
// its chain, or Failed.
func Of(err error) int {
	if err == nil {
		return OK
	}
	var e *Error
	if errors.As(err, &e) {
		return e.Code
	}
	return Failed
}

// Name is the stable, lower-case name of a code, for a machine-readable error
// ("usage", "not_found"). An unknown code is "failed".
func Name(code int) string {
	switch code {
	case OK:
		return "ok"
	case Usage:
		return "usage"
	case Auth:
		return "auth"
	case Forbidden:
		return "forbidden"
	case NotFound:
		return "not_found"
	case Refused:
		return "refused"
	case Unavailable:
		return "unavailable"
	default:
		return "failed"
	}
}

// Classify gives err the exit code of its class. An error that already
// carries a code keeps it, so a command can set Usage itself and classify
// everything else in one place.
func Classify(err error) error {
	if err == nil {
		return nil
	}
	var e *Error
	if errors.As(err, &e) {
		return err
	}
	return &Error{Code: classOf(err), Err: err}
}

func classOf(err error) int {
	if errors.Is(err, client.ErrAuthExpired) {
		return Auth
	}
	var api *client.APIError
	if errors.As(err, &api) {
		return statusClass(api.Status)
	}
	// *url.Error, which every failed http.Client call returns, is a net.Error.
	var netErr net.Error
	if errors.As(err, &netErr) || errors.Is(err, context.DeadlineExceeded) {
		return Unavailable
	}
	return Failed
}

func statusClass(status int) int {
	switch {
	case status == 401:
		return Auth
	case status == 403:
		return Forbidden
	case status == 404:
		return NotFound
	case status == 408, status == 429, status >= 500:
		return Unavailable
	case status >= 400:
		return Refused
	default:
		return Failed
	}
}
```

```go
package main

import (
	"github.com/spf13/cobra"

	"tinycld.org/cli/exitcode"
)

// ExitError carries the process exit code main should use. It is
// exitcode.Error: the type lives in an importable package so the command
// modules of packages can set a code too. Usage errors (bad flags/arguments)
// exit 2; a run that failed exits 1. An error that isn't an *ExitError also
// exits 1.
type ExitError = exitcode.Error

// Usage wraps err as an exit-2 error: the caller misused the command.
func Usage(err error) error { return exitcode.New(exitcode.Usage, err) }

// Failed wraps err as an exit-1 error: the command ran but did not succeed.
func Failed(err error) error { return exitcode.New(exitcode.Failed, err) }

// exitCode picks the process exit code for err: the code carried by an
// *ExitError anywhere in its chain, or 1 for any other non-nil error.
func exitCode(err error) int { return exitcode.Of(err) }
```

```go
import (
	"encoding/json"
	"fmt"
	"strings"

	"github.com/spf13/cobra"

	"tinycld.org/cli/exitcode"
	"tinycld.org/cli/output"

// runFunc is the body of a command. It gets the output options and the --yes
// answer already resolved, and returns a plain error: newCommand gives the
// error its exit code and prints the JSON error document.
type runFunc func(cmd *cobra.Command, args []string, o output.Options, yes bool) error

// argSpec names one positional argument.
type argSpec struct {
	Name string `json:"name"`
	Help string `json:"help"`
}

// commandDef is everything a command declares about itself. `books schema`
// prints it, so what an agent reads is what the command checks.
type commandDef struct {
	// Name is the command word. Use is built from it and Args.
	Name    string
	Short   string
	Long    string
	Aliases []string
	Args    []argSpec
	// Input is the JSON Schema of the --input document; nil when the command
	// reads none.
	Input map[string]any
	// Output is the JSON Schema of the document the command writes to stdout
	// with --json. Every command has one.
	Output map[string]any
}

// The annotations `books schema` reads back from the command tree.
const (
	annotationArgs   = "books.args"
	annotationInput  = "books.input"
	annotationOutput = "books.output"
)

// newCommand is the one constructor for a Books command that does work. It
// checks the argument count, resolves the output options, runs fn, and passes
// the result through report.
//
// Args is cobra.ArbitraryArgs on purpose: a cobra validator fails before RunE,
// so its error would get exit 1 and no JSON error document.
func newCommand(def commandDef, fn runFunc) *cobra.Command {
	if def.Output == nil {
		panic("books cli: command " + def.Name + " declares no output schema")
	}
	if def.Args == nil {
		def.Args = []argSpec{}
	}
	use := def.Name
	for _, a := range def.Args {
		use += " <" + a.Name + ">"
	}
	return &cobra.Command{
		Use:         use,
		Short:       def.Short,
		Long:        def.Long,
		Aliases:     def.Aliases,
		Args:        cobra.ArbitraryArgs,
		Annotations: schemaAnnotations(def.Args, def.Input, def.Output),
		RunE: func(cmd *cobra.Command, args []string) error {
			o, yes, err := output.FromCommand(cmd)
			if err != nil {
				return exitcode.New(exitcode.Usage, err)
			}
			if len(args) != len(def.Args) {
				return report(cmd, o, usagef("%s", argCountMessage(cmd, def.Args, len(args))))
			}
			return report(cmd, o, fn(cmd, args, o, yes))
		},
	}
}

// newGroup is a command that only holds other commands. Without a RunE cobra
// prints the help and exits 0 for `books acounts list`, so a typing mistake
// would look like success to a script. The group refuses an unknown word with
// a usage error.
func newGroup(name, short string) *cobra.Command {
	return &cobra.Command{
		Use:   name,
		Short: short,
		Args:  cobra.ArbitraryArgs,
		RunE: func(cmd *cobra.Command, args []string) error {
			if len(args) == 0 {
				return cmd.Help()
			}
			o, _, err := output.FromCommand(cmd)
			if err != nil {
				return exitcode.New(exitcode.Usage, err)
			}
			return report(cmd, o, usagef("unknown command %q for %q; commands: %s",
				args[0], cmd.CommandPath(), strings.Join(subcommandNames(cmd), ", ")))
		},
	}
}

// schemaAnnotations stores what `books schema` prints for a command. A
// command that newCommand cannot build (one with a variable argument count)
// sets them itself.
func schemaAnnotations(args []argSpec, input, output map[string]any) map[string]string {
	return map[string]string{
		annotationArgs:   mustJSON(args),
		annotationInput:  mustJSON(input),
		annotationOutput: mustJSON(output),
	}
}

func subcommandNames(cmd *cobra.Command) []string {
	names := []string{}
	for _, sub := range cmd.Commands() {
		if !sub.Hidden {
			names = append(names, sub.Name())
		}
	}
	return names
}

func argCountMessage(cmd *cobra.Command, want []argSpec, got int) string {
	if len(want) == 0 {
		return fmt.Sprintf("%q takes no arguments, got %d", cmd.CommandPath(), got)
	}
	names := make([]string, len(want))
	for i, a := range want {
		names[i] = "<" + a.Name + ">"
	}
	return fmt.Sprintf("%q takes %s, got %d arguments", cmd.CommandPath(), strings.Join(names, " "), got)
}

func mustJSON(v any) string {
	data, err := json.Marshal(v)
	if err != nil {
		panic("books cli: " + err.Error())
	}
	return string(data)
}
```

```go
package cli

import (
	"encoding/json"
	"errors"
	"fmt"
	"sort"
	"strings"

	"github.com/spf13/cobra"

	"tinycld.org/cli/client"
	"tinycld.org/cli/exitcode"
	"tinycld.org/cli/output"
)

// errorDoc is the one document a failed command writes to stdout with --json,
// so an agent that parses stdout always gets JSON: the result, or this.
type errorDoc struct {
	Error errorBody `json:"error"`
}

type errorBody struct {
	// Code is the name of the exit code: usage, auth, forbidden, not_found,
	// refused, unavailable or failed.
	Code string `json:"code"`
	// Exit is the process exit code.
	Exit int `json:"exit"`
	// Message is the server's own message for a refusal, the text a person
	// would see in the app.
	Message string `json:"message"`
	// Status is the HTTP status, when the server answered.
	Status int `json:"status,omitempty"`
	// Fields holds the server's message for each field it refused.
	Fields map[string]fieldError `json:"fields,omitempty"`
}

type fieldError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

// usagef is an error for a call that is wrong before anything is sent: exit 2.
func usagef(format string, a ...any) error {
	return exitcode.New(exitcode.Usage, fmt.Errorf(format, a...))
}

// report is the last step of every command. It gives err its exit code, adds
// the refused fields to the text a person reads, and with --json writes the
// error document to stdout. The shell prints "Error: …" on stderr and exits.
func report(cmd *cobra.Command, o output.Options, err error) error {
	if err == nil {
		return nil
	}
	body := bodyOf(err)
	classified := exitcode.Classify(withFields(err, body.Fields))
	if o.Format == output.JSON {
		enc := json.NewEncoder(cmd.OutOrStdout())
		enc.SetIndent("", "    ")
		if encErr := enc.Encode(errorDoc{Error: body}); encErr != nil {
			return errors.Join(classified, encErr)
		}
	}
	return classified
}

func bodyOf(err error) errorBody {
	code := exitcode.Of(exitcode.Classify(err))
	body := errorBody{Code: exitcode.Name(code), Exit: code, Message: err.Error()}
	var api *client.APIError
	if !errors.As(err, &api) {
		return body
	}
	body.Status = api.Status
	if api.Message != "" {
		body.Message = api.Message
	}
	body.Fields = refusedFields(api.Body)
	return body
}

// refusedFields reads PocketBase's field errors: {"data": {"<field>":
// {"code": "...", "message": "..."}}}. A body of another shape (a proxy's
// HTML page) has none.
func refusedFields(raw []byte) map[string]fieldError {
	var parsed struct {
		Data map[string]fieldError `json:"data"`
	}
	if json.Unmarshal(raw, &parsed) != nil || len(parsed.Data) == 0 {
		return nil
	}
	return parsed.Data
}

// fieldsError adds the refused fields to the message. "Failed to create
// record." alone does not say what to change.
type fieldsError struct {
	err    error
	fields map[string]fieldError
}

func (e *fieldsError) Unwrap() error { return e.err }

func (e *fieldsError) Error() string {
	names := make([]string, 0, len(e.fields))
	for name := range e.fields {
		names = append(names, name)
	}
	sort.Strings(names)
	parts := make([]string, len(names))
	for i, name := range names {
		parts[i] = name + ": " + e.fields[name].Message
	}
	return e.err.Error() + " (" + strings.Join(parts, "; ") + ")"
}

func withFields(err error, fields map[string]fieldError) error {
	if len(fields) == 0 {
		return err
	}
	return &fieldsError{err: err, fields: fields}
}
```

```go
package cli

import (
	"bytes"
	"encoding/json"
	"errors"
	"io"
	"os"

	"github.com/spf13/cobra"
)

// maxInputBytes bounds an --input document. The largest list record (a
// customer with two addresses and 5000 characters of notes) is far below it.
const maxInputBytes = 1 << 20

// bindInput adds the --input flag. Every command that takes a record takes it
// this way, and no other: one JSON object, from a file or from stdin.
func bindInput(cmd *cobra.Command, path *string) {
	cmd.Flags().StringVar(path, "input", "", "JSON file that holds the record, or - for standard input")
}

// readInput reads the --input document. Every failure is a usage error:
// nothing has been sent yet.
func readInput(cmd *cobra.Command, path string) (map[string]any, error) {
	if path == "" {
		return nil, usagef("--input is required: a JSON file, or - for standard input")
	}
	var src io.Reader
	if path == "-" {
		src = cmd.InOrStdin()
	} else {
		file, err := os.Open(path)
		if err != nil {
			return nil, usagef("--input: %v", err)
		}
		defer file.Close()
		src = file
	}
	data, err := io.ReadAll(io.LimitReader(src, maxInputBytes+1))
	if err != nil {
		return nil, usagef("--input: %v", err)
	}
	if len(data) > maxInputBytes {
		return nil, usagef("--input is larger than 1 MiB")
	}
	return parseObject(data)
}

// parseObject decodes one JSON object. Numbers stay json.Number, so integer
// cents are sent as written. A name given twice is refused: encoding/json
// would keep the last one and drop the other without a word.
func parseObject(data []byte) (map[string]any, error) {
	dec := json.NewDecoder(bytes.NewReader(data))
	dec.UseNumber()
	first, err := dec.Token()
	if errors.Is(err, io.EOF) {
		return nil, usagef("--input is empty")
	}
	if err != nil {
		return nil, usagef("--input is not JSON: %v", err)
	}
	if delim, ok := first.(json.Delim); !ok || delim != '{' {
		return nil, usagef("--input must be one JSON object")
	}
	out := map[string]any{}
	for dec.More() {
		keyToken, err := dec.Token()
		if err != nil {
			return nil, usagef("--input is not JSON: %v", err)
		}
		key, _ := keyToken.(string)
		if _, twice := out[key]; twice {
			return nil, usagef("--input names %q twice", key)
		}
		var value any
		if err := dec.Decode(&value); err != nil {
			return nil, usagef("--input is not JSON: %v", err)
		}
		out[key] = value
	}
	if _, err := dec.Token(); err != nil {
		return nil, usagef("--input is not JSON: %v", err)
	}
	if _, err := dec.Token(); !errors.Is(err, io.EOF) {
		return nil, usagef("--input must be one JSON object, with nothing after it")
	}
	return out, nil
}
```

```go
package cli

import (
	"bytes"
	_ "embed"
	"encoding/json"
	"fmt"
	"regexp"
	"slices"
)

// resources.json is the one definition of the list resources: which
// collection a command group reads, and each field's name, type and rules. The
// commands, the input checks and `books schema` are all built from it, and
// server/cli_resources_test.go checks it against the migrated collections, so
// a field that a migration renames fails a test, not a user.
//
//go:embed resources.json
var resourcesJSON []byte

// The field types of resources.json.
const (
	typeString  = "string"  // text, email, url or a one-value select
	typeBool    = "bool"    // true or false
	typeInteger = "integer" // a whole number (money is integer cents)
	typeNumber  = "number"  // a number that may have a fraction
	typeID      = "id"      // the id of one record of the resource named by ref
	typeIDs     = "ids"     // a list of ids of records of the resource named by ref
	typeAddress = "address" // an object with every key of addressKeys, or null
)

// addressKeys are the keys the server accepts in an address (server/address.go).
var addressKeys = []string{"line1", "line2", "city", "region", "postal_code", "country"}

// The fields every record has. PocketBase sets them.
var systemFields = []string{"id", "created", "updated"}

type field struct {
	Name string `json:"name"`
	Type string `json:"type"`
	// Required: the server refuses a record without a value.
	Required bool `json:"required,omitempty"`
	// ReadOnly: the server sets the value; the CLI refuses it in --input.
	ReadOnly bool `json:"readOnly,omitempty"`
	// Enum lists the values of a select.
	Enum []string `json:"enum,omitempty"`
	// Ref names the resource an id or ids field points to.
	Ref  string `json:"ref,omitempty"`
	Help string `json:"help"`
}

// filterFlag is a `list` flag that keeps the rows whose field equals the
// flag's value.
type filterFlag struct {
	Flag  string `json:"flag"`
	Field string `json:"field"`
	Help  string `json:"help"`
}

type resource struct {
	// Name is the command group: `tinycld books <name>`.
	Name string `json:"name"`
	// Singular is the word for one record, for messages.
	Singular   string `json:"singular"`
	Summary    string `json:"summary"`
	Collection string `json:"collection"`
	// Sort is the PocketBase sort of `list`.
	Sort string `json:"sort"`
	// Label is the field that names a record in messages. Empty: the id.
	Label string `json:"label"`
	// Columns are the fields of the `list` table. The id is always the last.
	Columns []string `json:"columns"`
	// Search are the fields `list --search` looks in.
	Search  []string     `json:"search"`
	Filters []filterFlag `json:"filters"`
	// Omit names the collection's fields the CLI leaves out on purpose. The
	// server test fails for a collection field that is neither in Fields nor
	// here, so a new column cannot be forgotten.
	Omit []string `json:"omit"`
	// DeleteHelp says when the server refuses a delete and what to do then.
	DeleteHelp string  `json:"deleteHelp"`
	Fields     []field `json:"fields"`
}

func (r *resource) field(name string) *field {
	for i := range r.Fields {
		if r.Fields[i].Name == name {
			return &r.Fields[i]
		}
	}
	return nil

// loadResources parses and checks the definitions. A mistake in the file is a
// programming error, found by the first test that loads the package.
func loadResources(data []byte) ([]resource, error) {
	var file struct {
		Resources []resource `json:"resources"`
	}
	dec := json.NewDecoder(bytes.NewReader(data))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&file); err != nil {
		return nil, err
	}
	names := map[string]bool{}
	for _, res := range file.Resources {
		names[res.Name] = true
	}
	seen := map[string]bool{}
	for i := range file.Resources {
		res := &file.Resources[i]
		if seen[res.Name] {
			return nil, fmt.Errorf("resource %q is defined twice", res.Name)
		}
		seen[res.Name] = true
		if err := checkResource(res, names); err != nil {
			return nil, fmt.Errorf("resource %q: %w", res.Name, err)
		}
	}
	return file.Resources, nil
}

func checkResource(res *resource, names map[string]bool) error {
	if res.Name == "" || res.Singular == "" || res.Summary == "" || res.Collection == "" || res.Sort == "" {
		return fmt.Errorf("name, singular, summary, collection and sort are required")
	}
	if len(res.Fields) == 0 {
		return fmt.Errorf("no fields")
	}
	fields := map[string]bool{}
	for i := range res.Fields {
		f := &res.Fields[i]
		if fields[f.Name] || slices.Contains(systemFields, f.Name) {
			return fmt.Errorf("field %q is defined twice or is a system field", f.Name)
		}
		fields[f.Name] = true
		if err := checkField(f, names); err != nil {
			return fmt.Errorf("field %q: %w", f.Name, err)
		}
	}
	uses := map[string][]string{"columns": res.Columns, "search": res.Search, "omit": nil}
	if res.Label != "" {
		uses["label"] = []string{res.Label}
	}
	for _, flt := range res.Filters {
		if flt.Flag == "" || flt.Help == "" {
			return fmt.Errorf("a filter needs a flag and a help text")
		}
		uses["filters"] = append(uses["filters"], flt.Field)
	}
	for where, list := range uses {
		for _, name := range list {
			if !fields[name] {
				return fmt.Errorf("%s names %q, which is not a field", where, name)
			}
		}
	}
	for _, name := range res.Search {
		if res.field(name).Type != typeString {
			return fmt.Errorf("search field %q is not a string", name)
		}
	}
	for _, name := range res.Omit {
		if fields[name] {
			return fmt.Errorf("omit names %q, which is a field", name)
		}
	}
	return nil
}

// fieldName is what a field name may be. Field names go into the filter and
// sort text of `list` as they are, so they are held to an identifier here;
// only values are bound.
var fieldName = regexp.MustCompile(`^[a-z][a-z0-9_]*$`)

func checkField(f *field, names map[string]bool) error {
	if !fieldName.MatchString(f.Name) {
		return fmt.Errorf("the name must be lower-case letters, digits and _")
	}
	if f.Help == "" {
		return fmt.Errorf("no help text")
	}
	switch f.Type {
	case typeString, typeBool, typeInteger, typeNumber, typeAddress:
		if f.Ref != "" {
			return fmt.Errorf("only an id or ids field has a ref")
		}
	case typeID, typeIDs:
		if !names[f.Ref] {
			return fmt.Errorf("ref %q is not a resource", f.Ref)
		}
	default:
		return fmt.Errorf("unknown type %q", f.Type)
	}
	if len(f.Enum) > 0 && f.Type != typeString {
		return fmt.Errorf("only a string field has an enum")
	}
	if f.Required && f.ReadOnly {
		return fmt.Errorf("a read-only field is set by the server and cannot be required of the caller")
	}
	return nil
}

// fieldSchema is the JSON Schema of one field's value.
func fieldSchema(f *field) map[string]any {
	schema := map[string]any{"description": f.Help}
	switch f.Type {
	case typeString:
		schema["type"] = "string"
		if len(f.Enum) > 0 {
			values := slices.Clone(f.Enum)
			if !f.Required {
				// A select with no value is "".
				values = append(values, "")
			}
			schema["enum"] = values
		}
	case typeBool:
		schema["type"] = "boolean"
	case typeInteger:
		schema["type"] = "integer"
	case typeNumber:
		schema["type"] = "number"
	case typeID:
		schema["type"] = "string"
		schema["description"] = f.Help + " (the id of a record of `books " + f.Ref + "`)"
	case typeIDs:
		schema["type"] = "array"
		schema["items"] = map[string]any{"type": "string"}
		schema["description"] = f.Help + " (ids of records of `books " + f.Ref + "`)"
	case typeAddress:
		parts := map[string]any{}
		for _, key := range addressKeys {
			parts[key] = map[string]any{"type": "string"}
		}
		schema["type"] = []string{"object", "null"}
		schema["properties"] = parts
		schema["required"] = addressKeys
		schema["additionalProperties"] = false
	}
	return schema
}

// recordSchema is the JSON Schema of a record as every command prints it: the
// id, the resource's fields in their declared order, created and updated.
func recordSchema(res *resource) map[string]any {
	props := map[string]any{
		"id":      map[string]any{"type": "string", "description": "the record id; the argument of show, edit, delete"},
		"created": map[string]any{"type": "string", "description": "when the record was created (UTC)"},
		"updated": map[string]any{"type": "string", "description": "when the record last changed (UTC)"},
	}
	for i := range res.Fields {
		props[res.Fields[i].Name] = fieldSchema(&res.Fields[i])
	}
	return map[string]any{
		"type":        "object",
		"description": "Every key is always present. A key that this server does not have (a server older than this CLI) is null.",
		"properties":  props,
		"required":    recordKeys(res),
	}
}

// listSchema is the JSON Schema of what `list` prints.
func listSchema(res *resource) map[string]any {
	return map[string]any{
		"type": "object",
		"properties": map[string]any{
			"items": map[string]any{"type": "array", "items": recordSchema(res)},
			"page":  map[string]any{"type": "integer", "description": "the page that was asked for, from 1"},
			"limit": map[string]any{"type": "integer", "description": "the page size that was asked for"},
			"total": map[string]any{"type": "integer", "description": "the number of records that match, on all pages"},
		},
		"required": []string{"items", "page", "limit", "total"},
	}
}

// recordKeys is the key order of a printed record.
func recordKeys(res *resource) []string {
	keys := []string{"id"}
	for i := range res.Fields {
		keys = append(keys, res.Fields[i].Name)
	}
	return append(keys, "created", "updated")
```

```go
var (
	reBoolClause   = regexp.MustCompile(`^(\w+) = (true|false)$`)
	reEqualClause  = regexp.MustCompile(`^(\w+) = "((?:[^"\\]|\\.)*)"$`)
	reSearchClause = regexp.MustCompile(`^(\w+) ~ "((?:[^"\\]|\\.)*)"$`)
)

func unquote(s string) string {
	return strings.NewReplacer(`\"`, `"`, `\\`, `\`).Replace(s)
}

// matches evaluates the filters listFilter builds: clauses joined by " && ",
// each `field = true|false`, `field = "text"`, or a group
// `(a ~ "text" || b ~ "text")`.
func (f *fakeBooks) matches(row rawRecord, filter string) bool {
	if filter == "" {
		return true
	}
	for _, clause := range strings.Split(filter, " && ") {
		switch {
		case reBoolClause.MatchString(clause):
			m := reBoolClause.FindStringSubmatch(clause)
			have, _ := row[m[1]].(bool)
			if have != (m[2] == "true") {
				return false
			}
		case reEqualClause.MatchString(clause):
			m := reEqualClause.FindStringSubmatch(clause)
			if have, _ := row[m[1]].(string); have != unquote(m[2]) {
				return false
			}
		case strings.HasPrefix(clause, "(") && strings.HasSuffix(clause, ")"):
			found := false
			for _, alternative := range strings.Split(clause[1:len(clause)-1], " || ") {
				m := reSearchClause.FindStringSubmatch(alternative)
				if m == nil {
					f.t.Errorf("unsupported filter clause %q in %q", alternative, filter)
					return false
				}
				have, _ := row[m[1]].(string)
				if strings.Contains(strings.ToLower(have), strings.ToLower(unquote(m[2]))) {
					found = true
				}
			}
			if !found {
				return false
			}
		default:
			f.t.Errorf("unsupported filter clause %q in %q", clause, filter)
			return false
		}
	}
	return true
}
```

```go
package cli

import (
	"bytes"
	"encoding/json"
	"sort"
	"strings"
)

// rawRecord is a record as the server sends it. Numbers stay json.Number, so
// integer cents are printed with the digits the server sent.
type rawRecord map[string]any

func (r *rawRecord) UnmarshalJSON(data []byte) error {
	dec := json.NewDecoder(bytes.NewReader(data))
	dec.UseNumber()
	var m map[string]any
	if err := dec.Decode(&m); err != nil {
		return err
	}
	*r = m
	return nil
}

// record is a record as the CLI prints it: exactly the keys of recordKeys, in
// that order. A key the server sends and resources.json does not name is
// dropped, and a key the server does not send is null, so the output of a
// command has one shape whatever version the server is.
type record struct {
	keys   []string
	values map[string]any
}

func project(res *resource, raw rawRecord) record {
	keys := recordKeys(res)
	values := make(map[string]any, len(keys))
	for _, key := range keys {
		values[key] = raw[key]
	}
	return record{keys: keys, values: values}
}

func (r record) MarshalJSON() ([]byte, error) {
	var buf bytes.Buffer
	buf.WriteByte('{')
	for i, key := range r.keys {
		if i > 0 {
			buf.WriteByte(',')
		}
		name, err := json.Marshal(key)
		if err != nil {
			return nil, err
		}
		value, err := json.Marshal(r.values[key])
		if err != nil {
			return nil, err
		}
		buf.Write(name)
		buf.WriteByte(':')
		buf.Write(value)
	}
	buf.WriteByte('}')
	return buf.Bytes(), nil
}

func (r record) id() string { return r.cell("id") }

// label names the record in a message: the resource's label field, or the id.
func (r record) label(res *resource) string {
	if res.Label != "" {
		if text := r.cell(res.Label); text != "" {
			return text
		}
	}
	return r.id()
}

// cell is a value as table text.
func (r record) cell(key string) string {
	return cellText(r.values[key])
}

func cellText(v any) string {
	switch t := v.(type) {
	case nil:
		return ""
	case string:
		return t
	case json.Number:
		return t.String()
	case bool:
		if t {
			return "yes"
		}
		return "no"
	case []any:
		parts := make([]string, len(t))
		for i, item := range t {
			parts[i] = cellText(item)
		}
		return strings.Join(parts, ",")
	case map[string]any:
		keys := make([]string, 0, len(t))
		for key := range t {
			keys = append(keys, key)
		}
		sort.Strings(keys)
		parts := []string{}
		for _, key := range keys {
			if text := cellText(t[key]); text != "" {
				parts = append(parts, text)
			}
		}
		return strings.Join(parts, ", ")
	default:
		data, _ := json.Marshal(t)
		return string(data)
	}
}
```

```go
import (
```

```go
type listResult struct {
	Items []record `json:"items"`
	Page  int      `json:"page"`
	Limit int      `json:"limit"`
	Total int      `json:"total"`
}

type listOptions struct {
	limit        int
	page         int
	search       string
	onlyInactive bool
	all          bool
	// filters holds the value of each filter flag, by flag name.
	filters map[string]*string
}

func newListCmd(c *client.Client, res *resource) *cobra.Command {
	opts := listOptions{filters: map[string]*string{}}
	long := "List " + res.Name + ", " + fmt.Sprint(defaultListLimit) + " to a page.\n\n" +
		"`total` in the --json result is the number of records that match on all\n" +
		"pages: ask for --page 2 while page × limit is below it."
	if res.hasInactive() {
		long += "\n\nInactive records are left out unless --inactive or --all is given."
	}
	cmd := newCommand(commandDef{
		Name:    "list",
		Short:   "List " + res.Name,
		Long:    long,
		Aliases: []string{"ls"},
		Output:  listSchema(res),
	}, func(cmd *cobra.Command, _ []string, o output.Options, _ bool) error {
		if opts.limit < 1 || opts.limit > maxListLimit {
			return usagef("--limit must be from 1 to %d", maxListLimit)
		}
		if opts.page < 1 {
			return usagef("--page must be 1 or more")
		}
		// Checked here and not with cobra's MarkFlagsMutuallyExclusive: cobra
		// fails before the command runs, with exit 1 and no JSON error document.
		if opts.onlyInactive && opts.all {
			return usagef("--inactive and --all cannot be used together")
		}
		page, err := client.ListRecords[rawRecord](cmd.Context(), c, res.Collection, client.ListOptions{
			Filter: listFilter(cmd, res, opts), Sort: res.Sort, Page: opts.page, PerPage: opts.limit,
		})
		if err != nil {
			return err
		}
		result := listResult{Items: make([]record, 0, len(page.Items)), Page: opts.page, Limit: opts.limit, Total: page.TotalItems}
		headers := make([]string, 0, len(res.Columns)+1)
		for _, column := range res.Columns {
			headers = append(headers, strings.ToUpper(column))
		}

		}

	})

	}

		fl.BoolVar(&opts.onlyInactive, "inactive", false, "list only the inactive "+res.Name)
		fl.BoolVar(&opts.all, "all", false, "list active and inactive "+res.Name)
	}
	for _, flt := range res.Filters {
		opts.filters[flt.Flag] = fl.String(flt.Flag, "", flt.Help)
	}
	return cmd
}

// listFilter builds the PocketBase filter. Every value a caller typed is
// bound through client.Filter, which quotes it: a quote in --search is text to
// look for, never filter syntax.
func listFilter(cmd *cobra.Command, res *resource, opts listOptions) string {
	clauses := []string{}
	params := map[string]any{}
	if res.hasInactive() && !opts.all {
		if opts.onlyInactive {
			clauses = append(clauses, "inactive = true")
		} else {
			clauses = append(clauses, "inactive = false")
		}
	}
	for i, flt := range res.Filters {
		if !cmd.Flags().Changed(flt.Flag) {
			continue
		}
		name := fmt.Sprintf("f%d", i)
		params[name] = *opts.filters[flt.Flag]
		clauses = append(clauses, flt.Field+" = {:"+name+"}")
	}
	if opts.search != "" && len(res.Search) > 0 {
		params["q"] = opts.search
		alternatives := make([]string, len(res.Search))
		for i, name := range res.Search {
			alternatives[i] = name + " ~ {:q}"
		}
		clauses = append(clauses, "("+strings.Join(alternatives, " || ")+")")
	}
	return client.Filter(strings.Join(clauses, " && "), params)
}
```

```go
package cli

import (
	"encoding/json"
	"slices"
	"strings"

	"github.com/spf13/cobra"

	"tinycld.org/cli/client"
	"tinycld.org/cli/output"
)

// inputSchema is the JSON Schema of the --input document of `add` (create)
// or `edit`. It is what checkInput enforces.
func inputSchema(res *resource, create bool) map[string]any {
	props := map[string]any{}
	required := []string{}
	for _, f := range res.writable() {
		prop := fieldSchema(f)
		if f.Required {
			required = append(required, f.Name)
		} else {
			allowNull(prop)
		}
		props[f.Name] = prop
	}
	schema := map[string]any{
		"type":                 "object",
		"properties":           props,
		"additionalProperties": false,
	}
	if create {
		schema["required"] = required
	} else {
		schema["minProperties"] = 1
	}
	return schema
}

// allowNull adds null to the schema of a field's value: in an input, null
// clears a field that is not required (checkValue).
func allowNull(schema map[string]any) {
	switch kind := schema["type"].(type) {
	case string:
		schema["type"] = []string{kind, "null"}
	case []string:
		if !slices.Contains(kind, "null") {
			schema["type"] = append(slices.Clone(kind), "null")
		}
	}
	if values, ok := schema["enum"].([]string); ok {
		withNull := make([]any, 0, len(values)+1)
		for _, value := range values {
			withNull = append(withNull, value)
		}
		schema["enum"] = append(withNull, nil)
	}
}

// checkInput refuses an --input document the server would refuse for its
// shape, before anything is sent, and names the mistake: an unknown field, a
// field the server sets, a value of the wrong JSON type, a value outside a
// select, a missing required field. Rules that need the data (a unique name, an
// active parent) stay with the server.
func checkInput(res *resource, body map[string]any, create bool) error {
	names := make([]string, 0, len(body))
	for name := range body {
		names = append(names, name)
	}
	slices.Sort(names)
	for _, name := range names {
		f := res.field(name)
		if f == nil {
			if slices.Contains(systemFields, name) {
				return usagef("%q is set by the server: leave it out of --input", name)
			}
			return usagef("%s has no field %q; fields: %s", res.Name, name, strings.Join(writableNames(res), ", "))
		}
		if f.ReadOnly {
			return usagef("%q is set by the server: leave it out of --input", name)
		}
		if err := checkValue(f, body[name]); err != nil {
			return err
		}
	}
	if !create {
		if len(body) == 0 {
			return usagef("nothing to change: --input is an empty object")
		}
		return nil
	}
	for _, f := range res.writable() {
		if _, given := body[f.Name]; f.Required && !given {
			return usagef("%q is required", f.Name)
		}
	}
	return nil
}

func writableNames(res *resource) []string {
	names := []string{}
	for _, f := range res.writable() {
		names = append(names, f.Name)
	}
	return names
}

// checkValue checks one value against its field's type. null clears a field
// that is not required.
func checkValue(f *field, value any) error {
	if value == nil {
		if f.Required {
			return usagef("%q is required: it cannot be null", f.Name)
		}
		return nil
	}
	switch f.Type {
	case typeString, typeID:
		text, ok := value.(string)
		if !ok {
			return usagef("%q must be a string", f.Name)
		}
		if f.Required && text == "" {
			return usagef("%q is required: it cannot be empty", f.Name)
		}
		if len(f.Enum) > 0 && text != "" && !slices.Contains(f.Enum, text) {
			return usagef("%q must be one of: %s", f.Name, strings.Join(f.Enum, ", "))
		}
	case typeBool:
		if _, ok := value.(bool); !ok {
			return usagef("%q must be true or false", f.Name)
		}
	case typeInteger:
		number, ok := value.(json.Number)
		if !ok {
			return usagef("%q must be a whole number", f.Name)
		}
		if _, err := number.Int64(); err != nil {
			return usagef("%q must be a whole number, without a fraction or an exponent", f.Name)
		}
	case typeNumber:
		if _, ok := value.(json.Number); !ok {
			return usagef("%q must be a number", f.Name)
		}
	case typeIDs:
		list, ok := value.([]any)
		if !ok {
			return usagef("%q must be an array of ids", f.Name)
		}
		for _, item := range list {
			if _, ok := item.(string); !ok {
				return usagef("%q must be an array of ids: each one a string", f.Name)
			}
		}
	case typeAddress:
		parts, ok := value.(map[string]any)
		if !ok {
			return usagef("%q must be an object with the keys %s, or null", f.Name, strings.Join(addressKeys, ", "))
		}
		for key, part := range parts {
			if !slices.Contains(addressKeys, key) {
				return usagef("%q has no key %q; keys: %s", f.Name, key, strings.Join(addressKeys, ", "))
			}
			if _, ok := part.(string); !ok {
				return usagef("%q: %q must be a string", f.Name, key)
			}
		}
		// The server takes an address only whole (server/address.go): a part
		// with no value is "". Its refusal does not name the missing key.
		for _, key := range addressKeys {
			if _, ok := parts[key]; !ok {
				return usagef("%q has no %q: an address names every key (%s); a part with no value is \"\"", f.Name, key, strings.Join(addressKeys, ", "))
			}
		}
	}
	return nil
}
```

```go
// deleteResult is what `delete` prints with --json.
type deleteResult struct {
	ID      string `json:"id"`
	Deleted bool   `json:"deleted"`
}

// newDeleteCmd deletes a record for good. Books has no trash: the server
// refuses to delete a record that transactions or other records use, and says
// so. The command asks first; without a terminal it needs --yes.
func newDeleteCmd(c *client.Client, res *resource) *cobra.Command {
	long := "Delete one " + res.Singular + ". This cannot be undone.\n\n" +
		"The command asks before it deletes. Without a terminal, pass --yes."
	if res.DeleteHelp != "" {
		long += "\n\n" + res.DeleteHelp
	}
	return newCommand(commandDef{
		Name:    "delete",
		Short:   "Delete one " + res.Singular + " for good",
		Long:    long,
		Aliases: []string{"rm", "remove"},
		Args:    idArg(res),
		Output: map[string]any{
			"type": "object",
			"properties": map[string]any{
				"id":      map[string]any{"type": "string", "description": "the id of the deleted record"},
				"deleted": map[string]any{"type": "boolean", "description": "true when the record was deleted; false when the question was answered with no"},
			},
			"required": []string{"id", "deleted"},
		},
	}, func(cmd *cobra.Command, args []string, o output.Options, yes bool) error {
		rec, err := fetchRecord(cmd.Context(), c, res, args[0])
		if err != nil {
			return err
		}
		question := fmt.Sprintf("Delete %s %q for good? This cannot be undone.", res.Singular, rec.label(res))
		ok, err := ui.Confirm(o, yes, cmd.InOrStdin(), cmd.ErrOrStderr(), question)
		if err != nil {
			// No terminal and no --yes, or an answer that could not be read:
			// the call is wrong, and nothing was deleted.
			return usagef("%v", err)
		}
		result := deleteResult{ID: rec.id(), Deleted: ok}
		if ok {
			if err := client.DeleteRecord(cmd.Context(), c, res.Collection, rec.id()); err != nil {
				return prefixID(res, rec.id(), err)
			}
			o.Info(cmd.ErrOrStderr(), "deleted %s %s (%s)", res.Singular, rec.label(res), rec.id())
		} else {
			o.Info(cmd.ErrOrStderr(), "not deleted")
		}
		return o.Write(cmd.OutOrStdout(), []string{"ID", "DELETED"},
			[][]string{{result.ID, cellText(result.Deleted)}}, result)
	})
}
```
